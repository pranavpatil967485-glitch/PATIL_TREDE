"-"} ${data.signal} `).join(""); } function startTradingBot() { if (botStatus) return; botStatus = true; document.getElementById("botStatus").innerText = "BOT STATUS: RUNNING (PAPER MODE)"; updateDashboard(); botTimer = setInterval( updateDashboard, 60000 ); } function stopTradingBot() { botStatus = false; clearInterval(botTimer); document.getElementById("botStatus").innerText = "BOT STATUS: STOPPED"; } updateDashboard();
